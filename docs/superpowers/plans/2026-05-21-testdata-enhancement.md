```sql
-- MIGRA-Go Test Data: v4 schema evolution
-- Version v4 extends v3 with IDENTITY columns, COLLATE, and FK CASCADE actions.
--
-- Changes from v3:
--   users.id:       SERIAL → INTEGER GENERATED ALWAYS AS IDENTITY
--   users.username:  added COLLATE "en_US"
--   posts.id:       SERIAL → INTEGER GENERATED BY DEFAULT AS IDENTITY
--   posts.user_id FK: added ON DELETE CASCADE

CREATE TABLE users (
    id INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
    username VARCHAR(50) NOT NULL COLLATE "en_US",
    email VARCHAR(100) DEFAULT 'unknown',
    created_at TIMESTAMP DEFAULT now(),
    phone VARCHAR(20)
);

CREATE TABLE posts (
    id INTEGER PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
    user_id INTEGER NOT NULL,
    title VARCHAR(200) NOT NULL,
    content TEXT,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE categories (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    sort_order INTEGER DEFAULT 0
);

CREATE INDEX idx_posts_user_id ON posts(user_id);
CREATE UNIQUE INDEX idx_users_username_unique ON users(username);

CREATE TYPE user_role AS ENUM ('admin', 'user', 'moderator');
```

```go
func TestV3UnsafeDropDiff_Safe(t *testing.T) {
	out, _, err := app.NewDiffService(newDefaultDeps()).Run(context.Background(), app.Config{
		Source:     "../../testdata/diff/v2/schema.sql",
		Target:     "../../testdata/diff/v3/schema.sql",
		Schemas:    []string{"public"},
		Format:     "sql",
		Timeout:    defaultDiffTimeout,
		UnsafeDrop: false, // safe mode
	})
	if err != nil {
		t.Fatalf("diff service failed: %v", err)
	}

	// Safe mode should NOT contain DROP statements
	for _, drop := range []string{"DROP COLUMN", "DROP TABLE", "DROP INDEX"} {
		if strings.Contains(out, drop) {
			t.Fatalf("safe mode should not contain %q, got:\n%s", drop, out)
		}
	}

	// But should contain non-destructive changes
	if !strings.Contains(out, `ADD COLUMN "phone"`) {
		t.Fatalf("expected ADD COLUMN phone, got:\n%s", out)
	}
}
```

```go
func TestParseAlterOperations(t *testing.T) {
	sql, err := os.ReadFile("../../testdata/alter_operations.sql")
	if err != nil {
		t.Fatalf("failed to read testdata: %v", err)
	}

	schema, err := ParseSQL(string(sql))
	if err != nil {
		t.Fatalf("ParseSQL failed: %v", err)
	}

	// Should parse items table successfully
	ns := schema.GetNamespace("public")
	if ns == nil {
		t.Fatal("expected public namespace")
	}
	items, ok := ns.Tables["items"]
	if !ok {
		t.Fatal("expected items table")
	}

	// Verify columns added via ALTER TABLE ADD COLUMN exist
	colNames := make(map[string]bool)
	for _, col := range items.Columns {
		colNames[col.Name] = true
	}
	for _, expected := range []string{"sku", "item_code", "tags", "sort_order", "uuid_col"} {
		if !colNames[expected] {
			t.Fatalf("expected column %q in items table", expected)
		}
	}

	t.Logf("Parsed %d columns from items table", len(items.Columns))
}
```
